CREATE PROCEDURE [dbo].HOLIDAY_TOGGLE_ABSENCE_TEAM
	@teamid integer
,	@date date
,	@type integer
,	@halfday bit
AS
	IF EXISTS( 	SELECT 1 
			FROM Absence 
			WHERE TeamID = @teamid 
			AND PersonID = NULL
			AND AbsenceDate = @date 
		)
	
		DELETE FROM Absence
		WHERE TeamID = @teamid 
		AND PersonID = NULL
		AND AbsenceDate = @date 
	
	ELSE
		INSERT INTO Absence (
		  PersonID
		, TeamID
		, AbsenceDate
		, AbsenceTypeID
		, HalfDay
		, WholeTeamAbsence
		) VALUES (
		  NULL
		, @teamid 
		, @date 
		, @type 
		, @halfday 
		, 1
		)
